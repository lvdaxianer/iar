```ts
test('service install/inspect remains tenant scoped and sidecar preserves safe errors', async () => {
  const api = createTenantServiceApi();
  const installed = await api.request(serviceRequest({ tenantId: 'tenant-a', actorId: 'actor-tenant-a', operation: 'plugin.install', payload: { manifest: manifest({ version: '1.0.0' }) } }));
  assert.equal(installed.error, null);
  const foreign = await api.request(serviceRequest({ tenantId: 'tenant-b', actorId: 'actor-tenant-b', operation: 'plugin.inspect', payload: { pluginId: 'plugin-a' } }));
  assert.equal(foreign.error?.code, 'plugin.unavailable');
  const sidecar = createSidecarTransport(api);
  const malformed = JSON.parse(await sidecar.handleJson('{"protocolVersion":"iar.sidecar/v0"}'));
  assert.equal(malformed.error.code, 'service.invalid_request');
  assert.equal(JSON.stringify(malformed).includes('api-key'), false);
  await api.close();
});

test('executor instances cannot cancel or clean up another plugin instance', async () => {
  const first = createWorkerExecutor({ version: '1.0.0', maxTimeoutMs: 100 });
  const second = createWorkerExecutor({ version: '1.0.0', maxTimeoutMs: 100 });
  const pending = first.execute(validRequest({ pluginId: 'plugin-a', requestId: 'a-1' }), async () => new Promise(() => undefined));
  await new Promise<void>((resolve) => setImmediate(resolve));
  assert.equal(second.cancel('a-1'), false);
  assert.equal(second.activeRequestCount(), 0);
  await first.close();
  await pending;
  await second.close();
});
```
